
extern void put_sprite_rle(int x, int y, unsigned char far *sprite);
#pragma aux put_sprite = \
    "push bp" \
    "mov  bp, sp" \
    "push si" \
    "push di" \
    "push ds" \
    "push es" \
    "push bx" \
    "push dx" \
    "push cx" \
    \
    /* load params */ \
    "mov bx, [bp+4]"          /* x */ \
    "mov ax, [bp+6]"          /* y */ \
    \
    /* y >>= 4  (sans SAR -> loop 4 fois SHR ax,1) */ \
    "shr ax, 1" \
    "shr ax, 1" \
    "shr ax, 1" \
    "shr ax, 1" \
    \
    "mov dx, ax"              /* y_shifted */ \
    \
    /* DS:SI = sprite */ \
    "mov si, [bp+8]" \
    "mov ax, [bp+10]" \
    "mov ds, ax" \
    \
    /* skip width,height */ \
    "add si, 2" \
    \
    /* Load buffer pointer */ \
    "mov ax, word ptr buffer" \
    "mov cx, word ptr buffer+2" \
    "mov es, cx" \
    \
"ps_outer:" \
    "mov al, [si]"            /* y_sprite */ \
    "cmp al, 255" \
    "je  ps_end" \
    \
    /* horizontal clip */ \
    "cmp bx, 0" \
    "jl  ps_end" \
    "cmp bx, 249" \
    "jg  ps_end" \
    \
    /* empty column ? */ \
    "cmp al, 254" \
    "jne ps_col" \
    \
    "inc bx" \
    "inc si" \
    "jmp ps_outer" \
    \
"ps_col:" \
    /* AL = y_sprite */ \
    "mov ah, al" \
    "inc si" \
    "mov dl, [si]"        /* dl = len */ \
    "inc si" \
    \
    /* start = y_shifted + y_sprite */ \
    "mov ax, dx" \
    "add ax, ah" \
    /* AX = start */ \
    \
    /* compute offset = start * 320 + x */ \
    /* AX = start */ \
    "mov cx, 320" \
    "mul cx"              /* DX:AX = AX * 320 */ \
    "add ax, bx" \
    "adc dx, 0" \
    \
    /* DI = offset = AX */ \
    "mov di, ax" \
    \
"ps_row:" \
    "cmp dl, 0" \
    "je  ps_next" \
    \
    /* vertical clipping : if start < 200 */ \
    "cmp ax, 200" \
    "jae ps_skip" \
    \
    /* write pixel */ \
    "mov al, [si]" \
    "mov es:[di], al" \
    \
"ps_skip:" \
    /* next line: offset += 320 */ \
    "add di, 320" \
    \
    "inc si" \
    "inc ax"          /* start++ */ \
    "dec dl" \
    "jmp ps_row" \
    \
"ps_next:" \
    "mov al, [si]" \
    "jmp ps_outer" \
    \
"ps_end:" \
    "pop cx" \
    "pop dx" \
    "pop bx" \
    "pop es" \
    "pop ds" \
    "pop di" \
    "pop si" \
    "mov sp, bp" \
    "pop bp" \
    "ret" \
    \
    parm [bx] [ax] [si] \
    modify [ax bx cx dx si di es];
